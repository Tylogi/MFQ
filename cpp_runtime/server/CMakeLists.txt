set(_version_file "${MFQ_REPOSITORY_ROOT}/mfq/_version.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_version_file}")
file(STRINGS "${_version_file}" _version_line REGEX "^__version__ = \"[^\"]+\"$")
if(NOT _version_line MATCHES "^__version__ = \"([^\"]+)\"$")
    message(FATAL_ERROR "Cannot read MFQ version from ${_version_file}")
endif()
set(_server_version "${CMAKE_MATCH_1}")

find_package(SQLite3 REQUIRED)
set(_schema_file "${MFQ_REPOSITORY_ROOT}/mfq/server/state/schema.sql")
set(_storage_file "${MFQ_REPOSITORY_ROOT}/mfq/server/state/storage.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_schema_file}" "${_storage_file}")
file(STRINGS "${_storage_file}" _schema_version_line REGEX "^SCHEMA_VERSION = [0-9]+$")
if(NOT _schema_version_line STREQUAL "SCHEMA_VERSION = 19")
    message(FATAL_ERROR "Update native SQLite migrations to match Python SCHEMA_VERSION")
endif()
file(READ "${_schema_file}" MFQ_SERVER_SCHEMA)
configure_file(storage/schema.h.in generated/schema.h @ONLY)

add_executable(mfq-server ../apps/server_main.cpp server.cpp instance.cpp
    auth/auth.cpp storage/storage.cpp storage/files.cpp models/catalog.cpp)
target_include_directories(mfq-server PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(mfq-server PRIVATE mfq-http mfq-json mfq-paged-prefix-cache mfq-model-source SQLite::SQLite3)
target_compile_definitions(mfq-server PRIVATE MFQ_SERVER_VERSION="${_server_version}")
if(WIN32)
    target_link_libraries(mfq-server PRIVATE bcrypt)
endif()
target_compile_features(mfq-server PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(mfq-server PRIVATE /utf-8 /EHsc)
endif()
if(MINGW)
    target_link_options(mfq-server PRIVATE -municode)
endif()

if(BUILD_TESTING)
    add_executable(mfq-server-test ../tests/server_test.cpp)
    target_include_directories(mfq-server-test PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(mfq-server-test PRIVATE mfq-http mfq-json SQLite::SQLite3)

    if(MSVC)

    else()
        target_compile_options(mfq-server-test PRIVATE -UNDEBUG)
    endif()
    if(MINGW)
        target_link_options(mfq-server-test PRIVATE -municode)
    endif()

    add_test(NAME mfq-server-test COMMAND mfq-server-test $<TARGET_FILE:mfq-server>)
    set_tests_properties(mfq-server-test PROPERTIES TIMEOUT 90 LABELS "native-server;stage1")
    add_test(NAME mfq-server-model-catalog-test COMMAND mfq-server-test $<TARGET_FILE:mfq-server> --models)
    set_tests_properties(mfq-server-model-catalog-test PROPERTIES TIMEOUT 90 LABELS "native-server;stage2")
endif()
