set(_version_file "${MFQ_REPOSITORY_ROOT}/mfq/_version.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_version_file}")
file(STRINGS "${_version_file}" _version_line REGEX "^__version__ = \"[^\"]+\"$")
if(NOT _version_line MATCHES "^__version__ = \"([^\"]+)\"$")
    message(FATAL_ERROR "Cannot read MFQ version from ${_version_file}")
endif()

add_executable(mfq-server ../apps/server_main.cpp server.cpp instance.cpp)
target_include_directories(mfq-server PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mfq-server PRIVATE mfq-http mfq-json mfq-paged-prefix-cache)
target_compile_definitions(mfq-server PRIVATE MFQ_SERVER_VERSION="${CMAKE_MATCH_1}")
target_compile_features(mfq-server PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(mfq-server PRIVATE /utf-8 /EHsc)
endif()
if(MINGW)
    target_link_options(mfq-server PRIVATE -municode)
endif()

if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME mfq-server-test COMMAND ${Python3_EXECUTABLE}
            ${PROJECT_SOURCE_DIR}/tests/server_process_test.py $<TARGET_FILE:mfq-server>)
        set_tests_properties(mfq-server-test PROPERTIES TIMEOUT 60)
    endif()
endif()
